\section{Descripción detallada de componentes del sistema SGH}

\subsection{Arquitectura General del Sistema}

El Sistema de Gestión de Horarios Académicos (SGH) está compuesto por cuatro componentes principales que interactúan a través de APIs REST:

\begin{enumerate}
\item \textbf{Aplicación Web (Next.js + Tailwind CSS)}: Interfaz principal para gestión administrativa, con páginas modulares y componentes reutilizables. Consume APIs REST para operaciones CRUD y visualización de datos.
\item \textbf{Backend (Spring Boot)}: Núcleo del sistema con arquitectura en capas (controladores, servicios, repositorios). Implementa lógica de negocio, algoritmos de optimización y persistencia de datos.
\item \textbf{Aplicación Móvil (React Native)}: Extensión móvil para consulta y gestión básica de horarios, con sincronización en tiempo real.
\item \textbf{Base de Datos (MySQL)}: Almacén relacional optimizado para consultas de horarios y entidades académicas.
\end{enumerate}

\subsection{Especificaciones Técnicas Detalladas}

\subsubsection{Aplicación Web}
\begin{itemize}
\item \textbf{Framework}: Next.js 14 con App Router
\item \textbf{Lenguaje}: TypeScript
\item \textbf{Estilos}: Tailwind CSS con configuración personalizada
\item \textbf{Estado}: React hooks + Context API
\item \textbf{Routing}: Next.js App Router con layouts anidados
\item \textbf{Componentes principales}:
  \begin{itemize}
  \item Dashboard: Panel con métricas y navegación
  \item CourseManager: Gestión CRUD de cursos
  \item ProfessorManager: Perfiles y disponibilidad
  \item ClassroomManager: Inventario de aulas
  \item ScheduleGenerator: Interfaz de optimización
  \item CalendarView: Visualización interactiva
  \end{itemize}
\end{itemize}

\subsubsection{Backend}
\begin{itemize}
\item \textbf{Framework}: Spring Boot 3.x
\item \textbf{Lenguaje}: Java 17
\item \textbf{Persistencia}: Spring Data JPA + Hibernate
\item \textbf{Seguridad}: Spring Security + JWT
\item \textbf{Documentación API}: OpenAPI/Swagger
\item \textbf{Controladores principales}:
  \begin{itemize}
  \item CourseController: Gestión de cursos
  \item ProfessorController: Gestión de profesores
  \item ClassroomController: Gestión de aulas
  \item ScheduleController: Generación y consulta de horarios
  \item UserController: Autenticación y perfiles
  \end{itemize}
\end{itemize}

\subsubsection{Base de Datos}
\begin{itemize}
\item \textbf{Motor}: MySQL 8.0
\item \textbf{Esquema principal}:
  \begin{itemize}
  \item \texttt{users}: Usuarios del sistema (roles: admin, professor, student)
  \item \texttt{courses}: Cursos académicos con prerrequisitos
  \item \texttt{professors}: Perfiles docentes con disponibilidad
  \item \texttt{classrooms}: Aulas con capacidad y equipamiento
  \item \texttt{schedules}: Asignaciones horario generadas
  \item \texttt{schedule\_entries}: Detalles de cada sesión
  \end{itemize}
\item \textbf{Índices optimizados}: Para consultas por fecha, profesor, curso y aula
\item \textbf{Migraciones}: Flyway para control de versiones del esquema
\end{itemize}

\subsubsection{Aplicación Móvil}
\begin{itemize}
\item \textbf{Framework}: React Native + Expo
\item \textbf{Navegación}: React Navigation
\item \textbf{Estado}: Context API + hooks personalizados
\item \textbf{Sincronización}: Axios para llamadas API
\item \textbf{Almacenamiento local}: AsyncStorage para cache offline
\item \textbf{Notificaciones}: Expo Notifications
\end{itemize}

\subsubsection{Ejemplo de Componente React - Gestión de Cursos}
\begin{verbatim}
// components/CourseManager.tsx
import React, { useState, useEffect } from 'react';
import { Course } from '../types/course';

interface CourseManagerProps {
  onCourseSelect: (course: Course) => void;
}

export const CourseManager: React.FC<CourseManagerProps> = ({
  onCourseSelect
}) => {
  const [courses, setCourses] = useState<Course[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchCourses();
  }, []);

  const fetchCourses = async () => {
    try {
      const response = await fetch('/api/courses');
      const data = await response.json();
      setCourses(data);
    } catch (error) {
      console.error('Error fetching courses:', error);
    } finally {
      setLoading(false);
    }
  };

  if (loading) return <div>Cargando cursos...</div>;

  return (
    <div className="course-manager">
      <h2>Gestión de Cursos</h2>
      <div className="course-list">
        {courses.map(course => (
          <div
            key={course.id}
            className="course-item"
            onClick={() => onCourseSelect(course)}
          >
            <h3>{course.name}</h3>
            <p>Código: {course.code}</p>
            <p>Créditos: {course.credits}</p>
          </div>
        ))}
      </div>
    </div>
  );
};
\end{verbatim}

\subsection{Guía de Despliegue}

\subsubsection{Prerrequisitos}
\begin{itemize}
\item Docker Desktop 4.0+
\item Node.js 18+
\item JDK 17+
\item MySQL 8.0+
\item 4GB RAM mínimo
\end{itemize}

\subsubsection{Pasos de Despliegue}
\begin{enumerate}
\item \textbf{Clonar repositorio}
\begin{verbatim}
git clone https://github.com/racinger/sgh.git
cd sgh
\end{verbatim}

\item \textbf{Configurar variables de entorno}
\begin{verbatim}
cp .env.example .env
# Editar .env con configuración específica
\end{verbatim}

\item \textbf{Construir imágenes Docker}
\begin{verbatim}
docker compose build
\end{verbatim}

\item \textbf{Iniciar servicios}
\begin{verbatim}
docker compose up -d
\end{verbatim}

\item \textbf{Verificar despliegue}
\begin{verbatim}
# Aplicación web: http://localhost:3000
# API Docs: http://localhost:8080/swagger-ui.html
# Base de datos: localhost:3306
\end{verbatim}
\end{enumerate}
